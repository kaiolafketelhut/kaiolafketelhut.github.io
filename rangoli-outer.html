import turtle, colorsys
t = turtle.Turtle()
turtle.Screen().bgcolor("black")
t.speed(0); t.pensize(2)
hsv = colorsys.hsv_to_rgb
for k in range(6,0,-1):
  r = k*45; n = 6+k*4
  for i in range(n):
    h = k/6 + i%2/8
    t.pencolor(hsv(h,.5,1))
    t.fillcolor(hsv(h,1,1))
    t.begin_fill()
    t.circle(r,60)
    t.left(120)
    t.circle(r,60)
    t.left(120)
    t.end_fill()
    t.left(360/n)
